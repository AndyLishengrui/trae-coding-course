输入第一行操作类型(S/M)。第二行0-11的行号。然后144个浮点数。
输出结果保留1位小数。
范围−10⁶到10⁶
输入
S
(144 numbers...)
输出
(sum, 1 decimal)
思路右上三角：j>i。双层循环i:0-11,j:0-11，满足j>i时累加。注意不包括对角线上元素。S求总和，M求平均除以66。
技巧右上三角=66个元素。S/M判断用if/else。
C++
#include <cstdio>

int main()
{
  char t;
  scanf("%c", &t);
  double a[12][12];

  for (int i = 0; i < 12; i ++ )
      for (int j = 0; j < 12; j ++ )
          scanf("%lf", &a[i][j]);

  int c = 0;
  double s = 0;

  for (int i = 0; i < 12; i ++ )
      for (int j = i + 1; j < 12; j ++ )
      {
        c ++ ;
        s += a[i][j];
      }

  if (t == 'S') printf("%.1lf\n", s);
  else printf("%.1lf\n", s / c);

  return 0;
}

NQ059平方矩阵IAcWing 753
输入整数N，输出N×N的回字形矩阵。每个位置的值=min(i+1,j+1,N-i,N-j)。
输入多个整数N(0结束)。
输出N×N矩阵，每个数占3字符宽。
范围1≤N≤100
输入
1
2
3
0
输出
  1

  1  1
  1  1

  1  1  1
  1  2  1
  1  1  1
思路数学公式法：val=min(i,j,n-1-i,n-1-j)+1(0-indexed)。每行空格分隔，每个数占3位。注意N=0结束。
技巧用数学公式替代模拟填充是理解抽象思维的好题。Python: f'{x:3d}'右对齐。
C++
#include <iostream>

using namespace std;

int main()
{
    int n;
    while (cin >> n, n)
    {
      for (int i = 1; i <= n; i ++ )
      {
        for (int j = 1; j <= n; j ++ )
        {
         int up = i, down = n - i + 1, left = j, right = n - j + 1;
         cout << min(min(up, down), min(left, right)) << ' ';
        }
        cout << endl;
      }

      cout << endl;
    }

    return 0;
}

C++
#include <iostream>

using namespace std;

int q[100][100];

int main()
{
    int n;
    while (cin >> n, n)
    {
     for (int i = 0; i < n; i ++ )
     {
       q[i][i] = 1;
       for (int j = i + 1, k = 2; j < n; j ++ , k ++ ) q[i][j] = k;
       for (int j = i + 1, k = 2; j < n; j ++ , k ++ ) q[j][i] = k;
     }

     for (int i = 0; i < n; i ++ )
     {
       for (int j = 0; j < n; j ++ ) cout << q[i][j] << ' ';
       cout << endl;
     }
     cout << endl;
    }

    return 0;
}

NQ066蛇形矩阵AcWing 756
输入n和m，输出n×m蛇形矩阵。从(1,1)开始，每次向右走到边界，向下走一步，再向左，向上……循环直到所有格填完。
输入一行两个整数n和m。
输出n行m列的蛇形矩阵。
范围1≤n,m≤100
输入
3 3
输出
1 2 3
8 9 4
7 6 5
思路方向数组+边界检测。d=[(0,1),(1,0),(0,-1),(-1,0)]依次控制→↓←↑。撞墙或已填则转向。while循环填充所有格子。
技巧方向数组是处理螺旋/蛇形遍历的标准模式。每次移动前检查是否越界或已填。
C++
#include <iostream>

using namespace std;

int res[100][100];

int main()
{
    int n, m;
    cin >> n >> m;

    int dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};

    for (int x = 0, y = 0, d = 0, k = 1; k <= n * m; k ++ )
    {
      res[x][y] = k;
      int a = x + dx[d], b = y + dy[d];
      if (a < 0 || a >= n || b < 0 || b >= m || res[a][b])
      {
        d = (d + 1) % 4;
        a = x + dx[d], b = y + dy[d];
      }
      x = a; y = b;
    }

    for (int i = 0; i < n; i ++ )
    {
      for (int j = 0; j < m; j ++ ) cout << res[i][j] << ' ';
      cout << endl;
    }

    return 0;
}
